function(tgx_add_example name)
    cmake_path(GET CMAKE_CURRENT_SOURCE_DIR FILENAME folder)
    set(target ${folder}_${name})
    add_executable(${target} ${name}.cpp)
    target_link_libraries(${target} PRIVATE tgx::tgx)
    target_compile_options(${target} PRIVATE ${TGX_WARNINGS})
endfunction()

add_subdirectory(tgx)
add_subdirectory(gl)
add_subdirectory(scenes)
add_subdirectory(games)
